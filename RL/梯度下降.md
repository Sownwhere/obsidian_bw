\begin{cases}
x_t = x_{t-1} - \alpha \left. \frac{\partial z}{\partial x} \right|_{x = x_{t-1}} \\
y_t = y_{t-1} - \alpha \left. \frac{\partial z}{\partial y} \right|_{y = y_{t-1}}
\end{cases}
$$
